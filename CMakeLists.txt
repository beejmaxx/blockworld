cmake_minimum_required(VERSION 3.30)
project(Blockworld VERSION 0.19.1 LANGUAGES C CXX)

if(NOT APPLE)
  message(FATAL_ERROR "Blockworld currently requires macOS (Metal and CoreText).")
endif()

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(APPLE)
  list(APPEND CMAKE_PREFIX_PATH /opt/homebrew /usr/local)
endif()
find_package(SDL3 3.4 REQUIRED CONFIG)
find_package(glm REQUIRED CONFIG)
find_package(Threads REQUIRED)

add_library(voxel_core src/world.cpp src/player.cpp src/adventure.cpp src/castle.cpp src/city.cpp src/coast.cpp src/harbor.cpp src/road.cpp src/countryside.cpp src/metropolis.cpp src/city_life.cpp src/minimap.cpp src/daylight.cpp src/sound.cpp src/crafting.cpp src/building.cpp src/farm.cpp src/inventory.cpp src/garden.cpp src/ranch.cpp)
target_include_directories(voxel_core PUBLIC src)
target_link_libraries(voxel_core PUBLIC glm::glm Threads::Threads)
target_compile_definitions(voxel_core PUBLIC GLM_FORCE_DEPTH_ZERO_TO_ONE)
target_compile_options(voxel_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(blockworld MACOSX_BUNDLE src/main.cpp src/renderer.cpp src/ui.cpp src/city_ui.cpp src/ui_font.cpp src/ui_font_native.c src/audio.cpp)
target_link_libraries(blockworld PRIVATE voxel_core SDL3::SDL3 "-framework CoreText" "-framework CoreGraphics" "-framework CoreFoundation")
target_compile_options(blockworld PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(blockworld PROPERTIES
  MACOSX_BUNDLE_BUNDLE_NAME Blockworld
  MACOSX_BUNDLE_GUI_IDENTIFIER dev.bijan.blockworld
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})

set(shader_sources shaders/world.metal shaders/ui.metal)
target_sources(blockworld PRIVATE ${shader_sources})
set_source_files_properties(${shader_sources} PROPERTIES
  HEADER_FILE_ONLY TRUE MACOSX_PACKAGE_LOCATION Resources/shaders)
